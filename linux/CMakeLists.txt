set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network)
find_package(KF6GlobalAccel REQUIRED)
find_package(KF6WindowSystem REQUIRED)
find_package(LayerShellQt REQUIRED)
add_executable(soy-scope main.cpp assets.qrc)
target_link_libraries(soy-scope PRIVATE Qt6::Widgets Qt6::Network KF6::GlobalAccel KF6::WindowSystem LayerShellQt::Interface)
install(TARGETS soy-scope DESTINATION bin)
add_executable(test-markers test-markers.cpp)
target_link_libraries(test-markers PRIVATE Qt6::Widgets)
add_test(NAME marker-state-machine COMMAND test-markers)
add_test(NAME transparent-image-mask COMMAND soy-scope --test-mask)

find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME setup-controls COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test-controls.py $<TARGET_FILE:soy-scope>)
install(FILES soy-scope.desktop DESTINATION share/applications)
install(FILES ../assets/soyscope-icon.png DESTINATION share/icons/hicolor/256x256/apps RENAME soy-scope.png)
